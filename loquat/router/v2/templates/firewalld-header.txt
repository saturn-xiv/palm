# https://firewalld.org/documentation/howto/enable-and-disable-firewalld.html
echo 'enable ipv4 forward'
cat >/etc/sysctl.d/100-router.conf <<EOF
net.ipv4.ip_forward = 1

net.ipv4.fib_multipath_hash_policy = 1
net.ipv4.fib_multipath_use_neigh = 1

net.ipv6.conf.all.disable_ipv6 = 1
net.ipv6.conf.default.disable_ipv6 = 1
net.ipv6.conf.lo.disable_ipv6 = 1
EOF
systemctl restart systemd-sysctl

nft flush ruleset
systemctl enable firewalld.service
systemctl restart firewalld
if [ -d /etc/firewalld/zones ]
then
    rm -r /etc/firewalld/zones
fi
mkdir -p /etc/firewalld/zones
if [ -f /etc/firewalld/direct.xml ]
then
    rm /etc/firewalld/direct.xml
fi
firewall-cmd --complete-reload
