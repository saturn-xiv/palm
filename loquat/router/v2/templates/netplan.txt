if [ -d /etc/netplan ]
then
    rm -r /etc/netplan
fi
mkdir /etc/netplan

{{ range $n, $p := range items }}
# {{ $p.label }}
cat >/etc/netplan/100-{{ $n }} <<EOF
{{ $p.content }}
EOF
{{ end }}

netplan apply

systemctl daemon-reload
systemctl restart systemd-networkd
systemctl restart systemd-resolved
systemctl restart systemd-timesyncd
